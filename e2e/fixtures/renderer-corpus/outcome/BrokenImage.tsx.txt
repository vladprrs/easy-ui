import { z } from "zod";
import type { EasyUIComponentProps } from "easy-ui/runtime";

/**
 * `outcome/`: **битое изображение**. Рядом стоят живой ассет (подставляется harness'ом) и
 * заведомо отсутствующий — id снова собирается конкатенацией, иначе publish 422-ит
 * непинуемый литерал.
 *
 * До R4 (§1.4) `settleImages` валил кадр только при **полном** отсутствии растра: один
 * загруженный `<img>` рядом с битым делал кадр «готовым».
 *
 * **Исход R4 (владелец ожиданий — R4, §6).** `decoded-strict` действительно выдаёт
 * `image_load_failed` с URL виновника — но только под политикой v2, то есть в профиле
 * `pixel-strict-v1` (N10: строгость приходит политикой, не флагом). Корпус снимает интерактивной
 * ручкой по политике v1, а readiness интерактивного пути advisory и статуса джобы не меняет,
 * поэтому ожидания фикстуры остаются `done`/`failureCode: null`: она фиксирует, что **дефолтный**
 * путь волной не поехал. Строгий исход доказывает `e2e/preview/capture-strictness.spec.ts`.
 */
export const definition = {
  props: z.strictObject({ caption: z.string() }),
  description: "Renderer corpus outcome fixture: a broken <img> next to a healthy asset",
  atomicLevel: "atom" as const,
  example: { caption: "broken-image" },
};

const MISSING_ASSET_ID = "asset_".concat("1".repeat(64));
const HEALTHY_URL = "/api/assets/__ASSET_PNG__";

export default function CorpusBrokenImage({ props }: EasyUIComponentProps<{ caption: string }>) {
  return (
    <div style={{ padding: 12, width: 320, fontFamily: '"YS Text", sans-serif' }}>
      <div style={{ fontSize: 11, fontWeight: 500, marginBottom: 6 }}>{props.caption}</div>
      <div style={{ display: "flex", gap: 10, alignItems: "flex-start" }}>
        <img src={HEALTHY_URL} alt="" width={64} height={64} />
        <img src={`/api/assets/${MISSING_ASSET_ID}`} alt="broken" width={64} height={64} />
      </div>
    </div>
  );
}
